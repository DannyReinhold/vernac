package operations;
import java.util.Optional;
import org.jspecify.annotations.Nullable;
import operations.domain.*;
import operations.domain.access.PlanningAccess;
public final class ExternalImplementation {
    public static Optional<Title> external(PlanningAccess self, @Nullable Title title) {
        if (title != null && title.string().equals("NULL")) return null;
        return Optional.ofNullable(title).map(self::decorate);
    }
}
