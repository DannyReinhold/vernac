package example.catalog;

import com.sun.net.httpserver.HttpServer;
import java.net.*;
import java.net.http.*;
import java.time.Duration;
import java.nio.charset.StandardCharsets;
import java.util.concurrent.atomic.*;
import example.catalog.domain.*;
import example.catalog.adapter.outbound.HttpCatalog;
import org.vernac.runtime.http.*;
import org.vernac.runtime.ports.*;
import org.vernac.runtime.json.*;
import static org.junit.jupiter.api.Assertions.*;

public final class Probe {
    public static void run() throws Exception {
        var status = new AtomicInteger(200);
        var body = new AtomicReference<>("{\"title\":\"Notebook\",\"added\":true}");
        var written = new AtomicReference<String>();
        var server = HttpServer.create(new InetSocketAddress("127.0.0.1", 0), 0);
        server.createContext("/catalog", exchange -> {
            try (exchange) {
                exchange.getResponseHeaders().set("Content-Type", "application/json");
                exchange.getResponseHeaders().set("X-Correlation-ID", "r-42");
                if (exchange.getRequestMethod().equals("PUT")) {
                    written.set(new String(exchange.getRequestBody().readAllBytes(), StandardCharsets.UTF_8));
                    exchange.sendResponseHeaders(204, -1);
                } else {
                    byte[] bytes = body.get().getBytes(StandardCharsets.UTF_8);
                    exchange.sendResponseHeaders(status.get(), bytes.length);
                    exchange.getResponseBody().write(bytes);
                }
            }
        });
        server.start();
        try (var transport = HttpClient.newHttpClient()) {
            var http = new HttpAdapterClient(transport, Duration.ofSeconds(3), 4096);
            CatalogSource catalog = new HttpCatalog(http, URI.create("http://127.0.0.1:" + server.getAddress().getPort()));
            assertEquals(Title.of("Notebook"), catalog.load());
            assertEquals(Title.of("Notebook"), catalog.lookup().orElseThrow());
            catalog.rename(Title.of("Renamed"));
            assertEquals("{\"title\":\"Renamed\"}", written.get());

            body.set("{\"title\":\" \"}");
            var invalidDomain = assertThrows(RemoteIntegrationException.class, catalog::load);
            assertInstanceOf(MappingContractException.class, invalidDomain.getCause());
            body.set("{}");
            var invalidDto = assertThrows(RemoteIntegrationException.class, catalog::load);
            assertEquals("$.title", ((DtoContractException)invalidDto.getCause()).path());
            body.set("{\"title\":42}");
            assertThrows(RemoteIntegrationException.class, catalog::load);

            status.set(404);
            assertTrue(catalog.lookup().isEmpty());
            assertThrows(PortInvocationException.class, catalog::load);
            status.set(403); body.set("{\"code\":\"ACCOUNT_BLOCKED\",\"ignored\":true}");
            var rejected = assertThrows(PortRejectedException.class, catalog::load);
            assertEquals(CatalogRejection.BLOCKED, rejected.code(CatalogRejection.class).orElseThrow());
            assertEquals("r-42", rejected.transportDetails().get("http.correlationId"));
            assertEquals("load", rejected.invocation().operation());
            status.set(401); body.set("{ malformed-secret");
            var failed = assertThrows(PortInvocationException.class, catalog::load);
            assertEquals(IntegrationCategory.AUTHENTICATION, failed.category());
            assertEquals("401", failed.transportDetails().get("http.status"));
            assertEquals(1, failed.getSuppressed().length);
            assertFalse(failed.getSuppressed()[0].toString().contains("malformed-secret"));
        } finally { server.stop(0); }
    }
}
