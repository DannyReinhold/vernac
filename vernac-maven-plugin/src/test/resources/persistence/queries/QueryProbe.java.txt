package querydemo;
import querydemo.domain.*;
import querydemo.adapter.outbound.jdbc.JdbcTourRepository;
import java.math.BigDecimal;
import java.time.Instant;
import javax.sql.DataSource;
import org.springframework.jdbc.core.namedparam.NamedParameterJdbcTemplate;
import org.springframework.jdbc.datasource.DataSourceTransactionManager;
import org.springframework.transaction.support.TransactionTemplate;
import org.springframework.transaction.IllegalTransactionStateException;
import org.vernac.runtime.NonUniqueQueryResultException;
import static org.junit.jupiter.api.Assertions.*;

public final class QueryProbe {
    public static void run(DataSource data) {
        var repo=new JdbcTourRepository(new NamedParameterJdbcTemplate(data));
        assertThrows(IllegalTransactionStateException.class,repo::all);
        var tx=new TransactionTemplate(new DataSourceTransactionManager(data));
        tx.executeWithoutResult(s->{
            assertTrue(repo.unique(Title.of("missing")).isEmpty());
            var stop=Stop.create(Title.of("Shared"));
            var a=Tour.create(Title.of("A"),null,Price.of(new BigDecimal("1.0")),Status.ACTIVE,Stops.of(stop,stop));
            var b=Tour.create(Title.of("B"),Title.of("R"),Price.of(new BigDecimal("1.00")),Status.CLOSED,Stops.of(Stop.reconstitute(stop.id(),Title.of("Other aggregate"))));
            repo.save(a); repo.save(b);
            assertEquals(2,repo.precedence(Title.of("A"),Title.of("B"),Status.CLOSED).size());
            assertEquals(b.id(),repo.grouped(Title.of("A"),Title.of("B"),Status.CLOSED).get(0).id());
            assertEquals(1,repo.grouped(Title.of("A"),Title.of("B"),Status.CLOSED).size());
            assertEquals(a.id(),repo.negated(Title.of("A")).get(0).id());
            assertEquals(0,repo.notReference(Title.of("R")).size());
            assertEquals(1,repo.notReference(Title.of("X")).size());
            assertEquals(2,repo.absentOrNot(Title.of("X")).size());
            assertEquals(b.id(),repo.booleanText("A%").get(0).id());
            var all=repo.all();
            assertEquals(2,all.size()); assertEquals("A",all.get(0).title().string());
            assertSame(all.get(0).stops().get(0),all.get(0).stops().get(1));
            assertNotSame(all.get(0).stops().get(0),all.get(1).stops().get(0));
            assertEquals("Other aggregate",all.get(1).stops().get(0).title().string());
            assertEquals(1,repo.priced(Price.of(new BigDecimal("1.0"))).size());
            assertEquals(2,repo.range(Price.of(BigDecimal.ONE),Price.of(BigDecimal.TEN)).size());
            assertEquals(1,repo.absent().size());
            assertEquals(0,repo.other(Title.of("R")).size()); // absent does not satisfy !=
            assertEquals(1,repo.other(Title.of("X")).size());
            assertEquals(1,repo.active(Status.ACTIVE).size());
            assertEquals(2,repo.before(Instant.now()).size());
            assertEquals(a.id(),repo.unique(Title.of("A")).orElseThrow().id());
            repo.save(Tour.create(Title.of("A"),null,Price.of(BigDecimal.ZERO),Status.ACTIVE,Stops.of()));
            assertThrows(NonUniqueQueryResultException.class,()->repo.unique(Title.of("A")));
            assertEquals(2,repo.titled(Title.of("A")).size());
        });
    }

    public static void pagination(DataSource data) {
        var jdbc=new NamedParameterJdbcTemplate(data);
        var repo=new JdbcTourRepository(jdbc);
        var tx=new TransactionTemplate(new DataSourceTransactionManager(data));
        var prefix="Page-"+java.util.UUID.randomUUID()+"-";
        tx.executeWithoutResult(s->{
            for(int i=0;i<5;i++) {
                var shared=Stop.create(Title.of("Shared"));
                repo.save(Tour.create(Title.of(prefix+i),null,Price.of(new BigDecimal("1.00")),Status.ACTIVE,Stops.of(shared,shared)));
            }
        });
        tx.executeWithoutResult(s->{
            var query=repo.counted(prefix);
            var first=query.page(org.vernac.runtime.pagination.OffsetRequest.ofPage(1,2));
            assertEquals(5,first.totalElements()); assertEquals(3,first.totalPages()); assertTrue(first.hasNext());
            assertSame(first.items().get(0).stops().get(0),first.items().get(0).stops().get(1));
            var beyond=query.page(org.vernac.runtime.pagination.OffsetRequest.ofPage(9,2));
            assertEquals(0,beyond.items().size()); assertEquals(5,beyond.totalElements()); assertEquals(9,beyond.pageNumber().orElseThrow());
            assertEquals(5,query.page(org.vernac.runtime.pagination.OffsetRequest.ofOffset(Long.MAX_VALUE,2)).totalElements());
            var unaligned=query.page(org.vernac.runtime.pagination.OffsetRequest.ofOffset(1,2));
            assertTrue(unaligned.pageNumber().isEmpty()); assertEquals(prefix+"1",unaligned.items().get(0).title().string());
            var empty=repo.counted(prefix+"missing").page(org.vernac.runtime.pagination.OffsetRequest.ofPage(1,2));
            assertEquals(0,empty.totalElements()); assertEquals(0,empty.totalPages()); assertFalse(empty.hasNext());
            var all=new java.util.ArrayList<Tour>();
            repo.paged(prefix).pages(org.vernac.runtime.pagination.OffsetRequest.ofPage(1,2)).items().forEach(all::add);
            assertEquals(5,all.size()); assertEquals(prefix+"4",all.getLast().title().string());
            var last=repo.paged(prefix).page(org.vernac.runtime.pagination.OffsetRequest.ofPage(3,2));
            assertEquals(1,last.items().size()); assertFalse(last.hasNext());
            var limited=new JdbcTourRepository(jdbc,2);
            assertThrows(org.vernac.runtime.DomainValidationException.class,()->limited.paged(prefix).page(org.vernac.runtime.pagination.OffsetRequest.ofPage(1,3)));
            assertEquals(2,limited.paged(prefix).page(org.vernac.runtime.pagination.OffsetRequest.ofPage(1,2)).items().size());
            assertEquals(5,new JdbcTourRepository(jdbc,Integer.MAX_VALUE).paged(prefix)
                    .page(org.vernac.runtime.pagination.OffsetRequest.ofPage(1,Integer.MAX_VALUE)).items().size());
        });
        var escaped=tx.execute(s->repo.counted(prefix));
        assertThrows(IllegalTransactionStateException.class,()->escaped.page(org.vernac.runtime.pagination.OffsetRequest.ofPage(1,2)));
        assertThrows(org.vernac.runtime.pagination.PaginationContractException.class,()->new JdbcTourRepository(jdbc,0));
        // Insert after root-window selection: the count must describe the original window, not a later count statement.
        var once=new java.util.concurrent.atomic.AtomicBoolean();
        var hooked=new NamedParameterJdbcTemplate(data) {
            @Override public <T> java.util.List<T> query(String sql,org.springframework.jdbc.core.namedparam.SqlParameterSource parameters,org.springframework.jdbc.core.RowMapper<T> mapper) {
                var rows=super.query(sql,parameters,mapper);
                if(sql.contains("@totalElements") && once.compareAndSet(false,true)) {
                    var other=new TransactionTemplate(new DataSourceTransactionManager(data));
                    other.setPropagationBehavior(org.springframework.transaction.TransactionDefinition.PROPAGATION_REQUIRES_NEW);
                    other.executeWithoutResult(s->repo.save(Tour.create(Title.of(prefix+"inserted"),null,Price.of(BigDecimal.ONE),Status.ACTIVE,Stops.of())));
                }
                return rows;
            }
        };
        tx.executeWithoutResult(s->{
            var page=new JdbcTourRepository(hooked).counted(prefix).page(org.vernac.runtime.pagination.OffsetRequest.ofPage(1,2));
            assertEquals(5,page.totalElements()); assertEquals(2,page.items().size());
            assertEquals(6,repo.counted(prefix).page(org.vernac.runtime.pagination.OffsetRequest.ofPage(1,2)).totalElements());
        });
        assertTrue(once.get());
    }

    public static void cursors(DataSource data) {
        var jdbc=new NamedParameterJdbcTemplate(data);
        var repo=new JdbcTourRepository(jdbc);
        var tx=new TransactionTemplate(new DataSourceTransactionManager(data));
        String prefix="Cursor-"+java.util.UUID.randomUUID()+"-";
        tx.executeWithoutResult(s->{
            var names=java.util.List.of("same","same","é","e\u0301","A","a","𐐀");
            for(int i=0;i<names.size();i++) {
                var shared=Stop.create(Title.of("Shared"));
                var instant=Instant.parse("2026-10-13T08:00:00Z").plusNanos(i);
                var amount=i==0?new BigDecimal("1.0"):i==1?new BigDecimal("1.00"):new BigDecimal(i);
                repo.save(Tour.reconstitute(TourId.create(),Title.of(prefix+names.get(i)),null,Price.of(amount),Status.ACTIVE,
                        Stops.of(shared,shared),instant,instant,0));
            }
        });
        tx.executeWithoutResult(s->{
            var expected=jdbc.queryForList("SELECT id FROM \"querydemo\".\"Tour\" WHERE title LIKE :p ORDER BY \"price\" DESC, title COLLATE \"C\" ASC, id DESC",
                    java.util.Map.of("p",prefix+"%"),java.util.UUID.class);
            var actual=new java.util.ArrayList<java.util.UUID>();
            repo.following(prefix).pages(org.vernac.runtime.pagination.CursorRequest.first(1)).items().forEach(t->{
                actual.add(t.id().value()); assertSame(t.stops().get(0),t.stops().get(1));
            });
            assertEquals(expected,actual); // includes equal-magnitude decimals with different scales
            var first=repo.following(prefix).page(org.vernac.runtime.pagination.CursorRequest.first(1));
            var token=first.nextCursor().orElseThrow();
            var second=repo.following(prefix).page(org.vernac.runtime.pagination.CursorRequest.after(token,3));
            assertEquals(expected.subList(1,4),second.items().stream().map(t->t.id().value()).toList());
            assertThrows(org.vernac.runtime.pagination.InvalidCursorException.class,()->repo.followingAgain(prefix).page(org.vernac.runtime.pagination.CursorRequest.after(token,1)));
            assertThrows(org.vernac.runtime.pagination.InvalidCursorException.class,()->repo.following(prefix+"wrong").page(org.vernac.runtime.pagination.CursorRequest.after(token,1)));
            String key=java.util.Base64.getEncoder().encodeToString(new byte[32]);
            var stable=new JdbcTourRepository(jdbc,1000,key).following(prefix).page(org.vernac.runtime.pagination.CursorRequest.first(1));
            var resumed=new JdbcTourRepository(jdbc,1000,key).following(prefix).page(stable.nextRequest().orElseThrow());
            assertEquals(expected.get(1),resumed.items().get(0).id().value());
            byte[] otherKey=new byte[32]; otherKey[0]=1;
            assertThrows(org.vernac.runtime.pagination.InvalidCursorException.class,()->new JdbcTourRepository(jdbc,1000,java.util.Base64.getEncoder().encodeToString(otherKey))
                    .following(prefix).page(stable.nextRequest().orElseThrow()));
            var decimalToken=repo.followingFiltered(prefix,Price.of(new BigDecimal("1.0"))).page(org.vernac.runtime.pagination.CursorRequest.first(1));
            assertEquals(1,decimalToken.items().size()); assertTrue(decimalToken.nextCursor().isEmpty());
            var expectedTime=jdbc.queryForList("SELECT id FROM \"querydemo\".\"Tour\" WHERE title LIKE :p ORDER BY \"@createdAt\" DESC, \"@nanoRemainder:@createdAt\" DESC, id ASC",
                    java.util.Map.of("p",prefix+"%"),java.util.UUID.class);
            var actualTime=new java.util.ArrayList<java.util.UUID>();
            repo.followingTime(prefix).pages(org.vernac.runtime.pagination.CursorRequest.first(2)).items().forEach(t->actualTime.add(t.id().value()));
            assertEquals(expectedTime,actualTime);
            var expectedString=jdbc.queryForList("SELECT id FROM \"querydemo\".\"Tour\" WHERE title LIKE :p ORDER BY title COLLATE \"C\" DESC, id ASC",
                    java.util.Map.of("p",prefix+"%"),java.util.UUID.class);
            var actualString=new java.util.ArrayList<java.util.UUID>();
            repo.followingString(prefix).pages(org.vernac.runtime.pagination.CursorRequest.first(1)).items().forEach(t->actualString.add(t.id().value()));
            assertEquals(expectedString,actualString);
            repo.delete(first.items().get(0));
            var afterDelete=repo.following(prefix).page(org.vernac.runtime.pagination.CursorRequest.after(token,20));
            assertEquals(expected.subList(1,expected.size()),afterDelete.items().stream().map(t->t.id().value()).toList());
            assertTrue(afterDelete.nextCursor().isEmpty());
            assertTrue(repo.following(prefix+"missing").page(org.vernac.runtime.pagination.CursorRequest.first(1)).items().isEmpty());
            assertThrows(org.vernac.runtime.DomainValidationException.class,()->new JdbcTourRepository(jdbc,1).following(prefix).page(org.vernac.runtime.pagination.CursorRequest.first(2)));
            assertThrows(org.vernac.runtime.pagination.InvalidCursorException.class,()->repo.following(prefix).page(org.vernac.runtime.pagination.CursorRequest.after(org.vernac.runtime.pagination.PageCursor.of("bad"),1)));
        });
        var escaped=tx.execute(s->repo.following(prefix));
        assertThrows(IllegalTransactionStateException.class,()->escaped.page(org.vernac.runtime.pagination.CursorRequest.first(1)));
    }
    public static void concurrent(DataSource data, boolean delete) { concurrent(data,delete,false); }
    public static void concurrent(DataSource data, boolean delete, boolean paginated) { concurrent(data,delete,paginated,false); }
    public static void concurrentCursor(DataSource data, boolean delete) { concurrent(data,delete,false,true); }
    private static void concurrent(DataSource data, boolean delete, boolean paginated, boolean cursor) {
        var manager=new DataSourceTransactionManager(data);
        var writer=new JdbcTourRepository(new NamedParameterJdbcTemplate(data));
        var tx=new TransactionTemplate(manager);
        var id=tx.execute(s->writer.save(Tour.create(Title.of("Concurrent"),null,Price.of(BigDecimal.ONE),Status.ACTIVE,Stops.of())).id());
        var once=new java.util.concurrent.atomic.AtomicBoolean();
        var jdbc=new NamedParameterJdbcTemplate(data) {
            @Override public <T> java.util.List<T> query(String sql,
                    org.springframework.jdbc.core.namedparam.SqlParameterSource parameters,
                    org.springframework.jdbc.core.RowMapper<T> mapper) {
                var rows=super.query(sql,parameters,mapper);
                if(sql.contains("ORDER BY") && once.compareAndSet(false,true)) {
                    var other=new TransactionTemplate(manager);
                    other.setPropagationBehavior(org.springframework.transaction.TransactionDefinition.PROPAGATION_REQUIRES_NEW);
                    other.executeWithoutResult(s->{
                        var tour=writer.byId(id);
                        if(delete) writer.delete(tour); else writer.save(tour);
                    });
                }
                return rows;
            }
        };
        var reader=new JdbcTourRepository(jdbc);
        assertThrows(org.springframework.dao.OptimisticLockingFailureException.class,
                ()->tx.execute(s->cursor
                        ? reader.followingString("Concurrent").page(org.vernac.runtime.pagination.CursorRequest.first(20))
                        : paginated ? reader.counted("Concurrent").page(org.vernac.runtime.pagination.OffsetRequest.ofPage(1,20))
                        : reader.all()));
        assertTrue(once.get());
    }
}
